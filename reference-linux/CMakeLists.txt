set(FASTTURBO_WINDOWS "${PROJECT_SOURCE_DIR}/reference-windows")
set(FASTTURBO_VKFFT "${PROJECT_SOURCE_DIR}/reference-macos/2DFastTurbo/ThirdParty/VkFFTSource/vkFFT")
include("${PROJECT_SOURCE_DIR}/python/native/cuda/core.cmake")
file(READ "${FASTTURBO_VKFFT}/vkFFT.h" vkfft_header)
string(REGEX MATCH "return ([0-9]+); //X.XX.XX" vkfft_version "${vkfft_header}")
if(NOT CMAKE_MATCH_1)
  message(FATAL_ERROR "Cannot identify vendored VkFFT version")
endif()
set(vkfft_version "${CMAKE_MATCH_1}")
add_executable(fastturbo_reference main.cpp validation.cpp)
target_include_directories(fastturbo_reference PRIVATE "${PROJECT_SOURCE_DIR}/python/native/windows/vendor")
target_compile_definitions(fastturbo_reference PRIVATE FASTTURBO_REFERENCE_ROOT="${PROJECT_SOURCE_DIR}")
target_compile_definitions(fastturbo_reference PRIVATE FASTTURBO_VKFFT_VERSION=${vkfft_version})
target_compile_options(fastturbo_reference PRIVATE -ffp-contract=off -Wall -Wextra -Wpedantic)
target_link_libraries(fastturbo_reference PRIVATE turbo_core cuda_platform)
set_target_properties(fastturbo_reference PROPERTIES BUILD_RPATH "${CUDAToolkit_LIBRARY_DIR}")
if(BUILD_TESTING)

  add_test(NAME reference_validation COMMAND fastturbo_reference --validate --json)
  add_test(NAME reference_benchmark COMMAND fastturbo_reference --n 32 --steps 7 --warmup 3 --benchmark --json)
  add_test(NAME reference_cli COMMAND "${CMAKE_COMMAND}"
    -DEXE=$<TARGET_FILE:fastturbo_reference> -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake")
  set_tests_properties(reference_validation reference_benchmark PROPERTIES TIMEOUT 180 RESOURCE_LOCK cuda)
endif()
