cmake_minimum_required(VERSION 3.26)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 86 CACHE STRING "CUDA architectures (86=RTX 3090, 90=H100)")
endif()
project(FastTurboReferenceLinux LANGUAGES CXX CUDA)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Use reference-windows or reference-macos on those platforms.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 20)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
include(CTest)
add_subdirectory(reference-linux)
option(FASTTURBO_BUILD_NATIVE_GUI "Build the native Linux Dear ImGui application" ON)
if(FASTTURBO_BUILD_NATIVE_GUI)
  add_subdirectory(native-linux)
endif()
